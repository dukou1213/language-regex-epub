#=================通用规则=================#
## 刪除多餘代碼 ##
    '</p><p>':r'</p>\n<p>',
    '</p>\n{2,}':r'</p>\n',
    '<p></p>':r'',
    '<a id=.*?></a>':r'',
    '</strong>\s<strong>':r' ',
    '<strong>\s</strong>':r' ',
## 主标题(根据需要改为h3或h2) ##
    '<body>\n+<h1>(.*?)</h1><p>':r'<body>\n<h3>\1</h3>\n<p>',
    '<body>\n+<p><strong>(.*)</strong></p>':r'<body>\n<h3>\1</h3>',
    '<body>\n+<p>(.*?)</p>':r'<body>\n<h3>\1</h3>',
## 脚注上标去掉空格 ##
    '<sup>(\[.\])\s</sup>':r'<sup>\1</sup> ',
## 转换雙引号 ##
    '&quot;':r'"',
## 日期 ##
    '<p>(.{1,18}\d+)\s*</p>\n</body>':r'<p class="date">\1</p>\n</body>',
    '<p>(.*\d.*(1|2)\d\d\d)</p>':r'<p class="date">\1</p>',
## 作者(用完停用此规则) ##
####'</(h3|h2)>\n<p>(.{0,80}?)</p>':r'</\1>\n<p class="author">\2</p>',
## 去除不规范的空格
    '<sup>\s+</sup>':r' ',
#=================祖鲁语ZU=================#
## 脚注 ##
    '<p>(Imibhalo\syaphansi:)\s*</p>\n<p>(.*)</p>\n<p>(.*)</p>\n</body>':r'<p class="footnote">\1</p>\n<p class="footnote">\2</p>\n<p class="footnote">\3</p>\n</body>',
    '<p>(Imibhalo\syaphansi:)\s*</p>\n<p>(.*)</p>\n</body>':r'<p class="footnote">\1</p>\n<p class="footnote">\2</p>\n</body>',
    '<p>(Imibhalo\syaphansi:)\s*</p>':r'<p class="footnote">\1</p>',
## 单独一行摘自 ##
    '<p>(Kucashunwe embhalweni othi.*)</p>':r'<p class="from">\1</p>',
    '<p>(kuvela embhalweni othi.*)</p>':r'<p class="from">\1</p>',
## 段落内摘自 ##
    '(”\s+)(?<!from-inline">)((\((?:\d+)?\s*[a-zA-ZéèëêîïôûüùçÉÈËÊÎÏÔÛÜÙÇ\s]+\s*\d+:\s?\d+(?:[-–]\d+)?(?:,\s*\d+)?\s*\)))':r'\1<span class="from-inline">\2</span>',
    '(</strong>”\s)(\(“[^<>]*?\)+)':r'\1<span class="from-inline">\2</span>',
## 参考圣经 ##
    '<p>(<strong>)?(Izikhombo Zamavesi EBhayibheli:)(</strong>)?</p>':r'<p class="heading-3">\2</p>',
## 相关神话语 ##
    '<p>(<strong>)?(Amazwi KaNkulunkulu Abalulekile:)(</strong>)?</p>':r'<p class="heading-3">\2</p>',
## 参考讲道交通 ##
    '<p>(<strong>)?(Izingcaphuno Zezintshumayelo Nokuhlanganyela Okungabhekiswa Kuzo:)(</strong>)?</p>':r'<p class="heading-3">\2</p>',
## 摘选全篇神话语 ##
    '<p>(\(Isiqephu Sezwi LikaNkulunkulu Esikhethiwe\))</p>\n<p><strong>(.*?)</strong></p>':r'<p class="heading-4">\1</p>\n<h4 class="sigil_not_in_toc">\2</h4>',
## 问题長標題 ##
    '<p><strong>(Umbuzo\s*\d:.*)</strong></p>':r'<h6>\1</h6>',
## 问题解答(另起新行) ##
    '<p>(<strong>)?(Impendulo:)(</strong>)?</p>':r'<p class="heading-3">\2</p>',
## 问题解答(與答案同一段) ##
    '<p><strong>(Impendulo:)(</strong>)?:\s*(</strong>)?\s*':r'<p><strong>\1:</strong> ',
## 规范span代码 ##
    '(<span class="from-inline">\()([^(]*?)\)\)</span>':r'\1\2)</span>)',
## 连续两个空格加代码 ##
    '(\d)\s\s':r'\1&#160;&#160;',
    '(\d)<strong>\s\s':r'\1&#160;&#160;<strong>',
}
