#=================通用规则=================#
## 刪除多餘代碼 ##
    '</p><p>':r'</p>\n<p>',
    '</p>\n{2,}':r'</p>\n',
    '<p></p>':r'',
    '<a id=.*?></a>':r'',
    '</strong>\s<strong>':r' ',
    '<strong>\s</strong>':r' ',
## 主标题(根据需要改为h3或h2) ##
    '<body>\n+<h1>(.*?)</h1><p>':r'<body>\n<h3>\1</h3>\n<p>',
    '<body>\n+<p><strong>(.*)</strong></p>':r'<body>\n<h3>\1</h3>',
    '<body>\n+<p>(.*?)</p>':r'<body>\n<h3>\1</h3>',
## 脚注上标去掉空格 ##
    '<sup>(\[.\])\s</sup>':r'<sup>\1</sup> ',
## 转换雙引号 ##
    '&quot;':r'"',
## 日期 ##
    '<p>(.{1,18}\d+)\s*</p>\n</body>':r'<p class="date">\1</p>\n</body>',
    '<p>(.*\d.*(1|2)\d\d\d)</p>':r'<p class="date">\1</p>',
    '<p>(.{0,10}(19|20)\d\d.{0,10})</p>':r'<p class="date">\1</p>',
## 作者(用完停用此规则) ##
####'</(h3|h2)>\n<p>(.{0,80}?)</p>':r'</\1>\n<p class="author">\2</p>',
## 去除不规范的空格
    '<sup>\s+</sup>':r' ',
#=================韩语KR=================#
## 脚注 ##
    '<p>(────+)</p>':r'<p class="footnote">\1</p>',
## 单独一行摘自 ##
    '<p>(―＜말씀이.*중에서)':r'<p class="from">\1',
    '<p>(―＜말씀.*중에서)':r'<p class="from">\1',
    '<p>(―＜말씀ㆍ\d권.*)</p>':r'<p class="from">\1</p>',
## 段落内摘自 ##
    '([”’])(?<!from-inline">)(\([^<>]{0,20}?\d+[\d:,-~]{0,20}\))':r'\1<span class="from-inline">\2</span>',
    '((?:</strong>)?[”』])(\(＜말씀ㆍ\d.*?＞ 중에서\))':r'\1<span class="from-inline">\2</span>',
    '(”)(?<!from-inline">)(\(.{0,20}중국어성경 직역\))':r'\1<span class="from-inline">\2</span>',
#    '(</strong>』)(\([^<>]*?\)+)':r'\1<span class="from-inline">\2</span>',
    '(”)(\(계 2, 3장\))':r'\1<span class="from-inline">\2</span>',
## 韩语《听神声音 看见神显现》 ##
    '<p><strong>(질문\(\d+\).*)</strong></p>':r'<h6>\1</h6>',
    '<p>(\(하나님 말씀 전편 발췌\))</p>\n<p><strong>(.*)</strong></p>':r'<p class="heading-4">\1</p>\n<h4 class="sigil_not_in_toc">\2</h4>',
    '<p><strong>(성경 참고:)</strong></p>':r'<p class="heading-3">\1</p>',
    '<p><strong>(관련된 하나님 말씀:)</strong></p>':r'<p class="heading-3">\1</p>',
    '<p><strong>(답변:)</strong></p>':r'<p class="heading-3">\1</p>',
### 每日神話 ##
    '<p>(?:<strong>)?(매일의 하나님 말씀)\s*(?:</strong>)?\s*(\(발췌문.*?\))(?:</strong>)?</p>':r'<p class="heading-1">\1<span class="excerpt">\2</span></p>',
    '<p><strong>(.*?)</strong>(\(발췌문\))</p>':r'<p class="heading-2">\1<span class="excerpt-small">\2</span></p>',
## 转换特殊符号 ##
    '&lt;':r'<',
    '&gt;':r'>',
## 文中脚注代码加空行 ##
#    '(<p class="footnote">[^\n]*?</p>)\n(?!<p\sclass="footnote"|<br|</body|<p>1|<p>a)':r'\1\n<br class="epub_blank" />\n',
## 规范span代码 ##
    '(<span class="from-inline">\()([^(]*?)\)\)</span>':r'\1\2)</span>)',
## 修改标签 ##
    '\.</span>':r'</span>.',
## 连续两个空格加代码 ##
    '(\d)\s\s':r'\1&#160;&#160;',
    '(\d)<strong>\s\s':r'\1&#160;&#160;<strong>',
## 卷三选段 ##
    '<p><strong>(발췌문..{0,5})</strong></p>':r'<p class="heading-1">\1</p>',
## 信神必须进入的真理实际 ##
    '<p>(?:<strong>)?(성경에 나오는 말|성경에 나오는 하나님의 말씀|말세의 전능하신 하나님의 말씀|관련 영화|관련 영화 발췌본|관련 동영상|관련 체험 간증|관련 설교|관련 찬양)\s*(?:</strong>)?</p>':r'<p class="heading-3">\1</p>',
