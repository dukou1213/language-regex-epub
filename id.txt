#=================通用规则=================#
## 刪除多餘代碼 ##
    '</p><p>':r'</p>\n<p>',
    '</p>\n{2,}':r'</p>\n',
    '<p></p>':r'',
    '<a id=.*?></a>':r'',
    '</strong>\s<strong>':r' ',
    '<strong>\s</strong>':r' ',
## 主标题(根据需要改为h3或h2) ##
    '<body>\n+<h1>(.*?)</h1><p>':r'<body>\n<h3>\1</h3>\n<p>',
    '<body>\n+<p><strong>(.*)</strong></p>':r'<body>\n<h3>\1</h3>',
    '<body>\n+<p>(.*?)</p>':r'<body>\n<h3>\1</h3>',
## 脚注上标去掉空格 ##
    '<sup>(\[.\])\s</sup>':r'<sup>\1</sup> ',
## 转换雙引号 ##
    '&quot;':r'"',
## 日期 ##
    '<p>(.{1,18}\d+)\s*</p>\n</body>':r'<p class="date">\1</p>\n</body>',
    '<p>(.*\d.*(1|2)\d\d\d)</p>':r'<p class="date">\1</p>',
## 作者(用完停用此规则) ##
####'</(h3|h2)>\n<p>(.{0,80}?)</p>':r'</\1>\n<p class="author">\2</p>',
## 去除不规范的空格
    '<sup>\s+</sup>':r' ',
#=================印尼语ID================#
## 脚注 ##
    '<p>(Catatan kaki:)\s*</p>\n<p>(.*)</p>\n<p>(.*)</p>\n</body>':r'<p class="footnote">\1</p>\n<p class="footnote">\2</p>\n<p class="footnote">\3</p>\n</body>',
    '<p>(Catatan kaki:)\s*</p>\n<p>(.*)</p>\n</body>':r'<p class="footnote">\1</p>\n<p class="footnote">\2</p>\n</body>',
    '<p>(Catatan kaki:)\s*</p>':r'<p class="footnote">\1</p>',
## 单独一行摘自 ##
    '<p>(Dikutip\sdari.*)</p>':r'<p class="from">\1</p>',
    '<p>(dari\s".*)</p>':r'<p class="from">\1</p>',
    '<p>(—Firman, Jilid \d.*)</p>':r'<p class="from">\1</p>',
## 段落内摘自 ##
    '(?<!from-inline">)((\((?:\d+)?\s*[a-zA-ZéèëêîïôûüùçÉÈËÊÎÏÔÛÜÙÇ\s]+\s*\d+:\s?\d+(?:[-–]\d+)?(?:,\s*\d+)?\s*\)))':r'<span class="from-inline">\1</span>',
    '((?:”|")\s+)(?<!from-inline">)(\([^<>]{1,20}?\d+[\d:-]*?\)+|\[[^<>]{1,20}?\d+[\d:-]*?\])':r'\1<span class="from-inline">\2</span>',
    '((?:</strong>)?”\s+)(?<!from-inline">)(\(Firman,\s(?:Vol\.|Jilid)[^<>]*?\)+)(\.)':r'\1<span class="from-inline">\2</span>\3',
## 参考圣经 ##
    '<p>(?:<strong>)?(Ayat\sAlkitab\suntuk\sReferensi:)(?:</strong>)?</p>':r'<p class="heading-3">\1</p>',

    '<p>(?:<strong>)?(Firman\sTuhan\syang\sRelevan:)(?:</strong>)?</p>':r'<p class="heading-3">\1</p>',
## 参考讲道交通，相关神话语 ##
    '<p><strong>(Kutipan\sKhotbah\sdan\sPersekutuan\suntuk\sReferensi:)</strong></p>':r'<p class="heading-3">\1</p>',
## 摘选全篇神话语 ##
    '<p>(?:<strong>)?(.*?)(?:</strong>)?</p>\n<p>(?:<strong>)?(\(Bab\sPilihan\sdari\sFirman\sTuhan\))(?:</strong>)?</p>':r'<h4 class="sigil_not_in_toc">\1</h4>\n<p class="heading-4">\2</p>',
## 问题長標題 ##
    '<p><strong>(Pertanyaan\s*\d:.*)</strong></p>':r'<h6>\1</h6>',
## 问题解答(另起新行) ##
    '<p>(?:<strong>)?(Jawaban:)(?:</strong>)?</p>':r'<p class="heading-3">\1</p>',

    '\.</span>':r'</span>.',
